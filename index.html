// REEMPLAZAR EL BLOQUE 'DOMContentLoaded' POR ESTE: window.addEventListener('DOMContentLoaded', () => { const savedToken = localStorage.getItem('googleToken'); const savedCustomUser = localStorage.getItem('customUser'); const savedTime = localStorage.getItem('tokenTimestamp'); // 60 minutos exactos (3,600,000 ms) para coincidir con la vigencia real de Google OAuth if (savedTime && (Date.now() - Number(savedTime) < 60 * 60 * 1000)) { if (savedToken) { googleToken = savedToken; try { const base64Url = googleToken.split('.')[1]; const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/'); const payload = JSON.parse(atob(base64)); // Verificar adicionalmente que el token de Google no haya expirado por reloj interno let ahora = Math.floor(Date.now() / 1000); if (payload.exp && payload.exp < ahora) { cerrarSesion(false); return; } activarSesionUI(payload.email); return; } catch (e) { cerrarSesion(false); } } else if (savedCustomUser) { customUser = savedCustomUser; activarSesionUI(savedCustomUser + ' (Solo lectura)'); return; } } cerrarSesion(false); });
